```python
#!/usr/bin/env python3
"""
V4.9.0.2 production scorer — V4901底座 + 风控增强 + 换手优化

与V4901的区别:
  - EMA平滑 0.7→0.6 (更稳定, 降换手)
  - 持仓保留bonus 0.2→0.3 (减少不必要调仓)
  - Market gate 0.30→0.35 (弱市更早减仓)
  - 个股CVaR止损降权 (近20日CVaR>5%的股票score×0.7)
"""

import numpy as np
import pandas as pd
import sqlite3
import logging
from pathlib import Path
from typing import Dict, List

from .v4901_production_scorer import V4901ProductionScorer

logger = logging.getLogger(__name__)

PROJECT_ROOT = Path(__file__).resolve().parent.parent.parent
DB_PATH = PROJECT_ROOT / 'data_adapter' / 'stock_data.db'


class V4902ProductionScorer(V4901ProductionScorer):
    """V4.9.0.2 scorer — 风控增强 + 换手优化"""

    # 覆盖参数
    EMA_ALPHA = 0.6          # V4901: 0.7 → 更平滑
    RETENTION_BONUS = 0.3    # V4901: 0.2 → 持仓惯性更大
    GATE_DONT_BUY = 0.35     # V4901: 0.30 → 弱市更早停买
    CVAR_PENALTY_THRESHOLD = 0.05  # 20日CVaR > 5% → 降权
    CVAR_PENALTY_FACTOR = 0.7      # 降权30%

    def __init__(self, model_type: str = 'small_data'):
        super().__init__(model_type=model_type)
        self._recent_returns_cache = {}

    def predict_scores(self, stock_codes: List[str], date: str) -> Dict[str, Dict]:
        """V4902: V4901 pipeline + CVaR止损降权"""
        results = super().predict_scores(stock_codes, date)

        # 加载近20日收益数据
        recent_returns = self._load_recent_returns(stock_codes, date)

        # CVaR止损降权
        for code, data in results.items():
            if code in recent_returns:
                ret_series = recent_returns[code]
                if len(ret_series) >= 20:
                    cvar = self._compute_cvar_simple(ret_series)
                    if cvar > self.CVAR_PENALTY_THRESHOLD:
                        penalty = self.CVAR_PENALTY_FACTOR
                        data['composite'] = data.get('composite', 0) * penalty
                        data['rank_score'] = data.get('rank_score', 0) * penalty
                        data['cvar_penalty'] = True
                        logger.debug(f"  {code} CVaR={cvar:.3f} > {self.CVAR_PENALTY_THRESHOLD} → score×{penalty}")

        # 重新计算全局百分位排名
        all_comp = [(code, data.get('composite', 0)) for code, data in results.items()]
        if all_comp:
            sorted_comp = sorted(all_comp, key=lambda x: x[1])
            n = len(sorted_comp)
            for rank_i, (code, _) in enumerate(sorted_comp):
                results[code]['score'] = round(rank_i / max(n - 1, 1) * 100, 1)

        return results

    def _load_recent_returns(self, stock_codes: List[str], date: str) -> Dict[str, np.ndarray]:
        """加载各股票近20日的price_change_pct"""
        date_str = date.replace('-', '')
        try:
            conn = sqlite3.connect(str(DB_PATH), timeout=30)
            placeholders = ','.join('?' * len(stock_codes))
            query = f"""
                SELECT s.code, dq.price_change_pct
                FROM daily_quotes dq
                JOIN securities s ON dq.security_id = s.id
                WHERE s.code IN ({placeholders})
                  AND dq.trade_date <= ?
                ORDER BY s.code, dq.trade_date DESC
            """
            df = pd.read_sql(query, conn, params=list(stock_codes) + [date_str])
            conn.close()

            result = {}
            for code, group in df.groupby('code'):
                # 取最近20条
                returns = group['price_change_pct'].head(20).values
                if len(returns) >= 10:
                    result[code] = returns
            return result
        except Exception as e:
            logger.warning(f"Failed to load recent returns: {e}")
            return {}

    @staticmethod
    def _compute_cvar_simple(returns: np.ndarray, alpha: float = 0.05) -> float:
        """简单CVaR计算 (无需pandas)"""
        sorted_r = np.sort(returns)
        n_tail = max(1, int(len(sorted_r) * alpha))
        return float(-sorted_r[:n_tail].mean())
```

```python
class V4902Trainer(V4901Trainer):
    """V4.9.0.2 训练器 — V4.9.0.1底座 + 风控增强训练

    改进:
    1. Sharpe-Blend 提高: 3d=0.15→0.25, 5d=0.30→0.45, 10d=0.40→0.50, 15d=0.40→0.50
       (基于V5.1诊断: CVaR=13.6%, MaxDD=-25.9%需降低)
    2. 下行样本加权×1.5: 让模型更关注避亏
    3. 训练结束时自动保存WF摘要JSON (WFER/OOS IC半衰期)
    """

    # 提高Sharpe-Blend (从V472的默认值按~1.5x缩放)
    TARGET_SHARPE_BLEND = {
        'label_3d':  0.25,   # V4901: 0.10 (via V472)
        'label_5d':  0.45,   # V4901: 0.25
        'label_10d': 0.50,   # V4901: 0.35
        'label_15d': 0.50,   # V4901: 0.35
    }

    DOWNSIDE_WEIGHT = 1.5  # 下行样本权重倍数

    def compute_sample_weights(self, df: pd.DataFrame, y: np.ndarray) -> np.ndarray:
        """V4.9.0.2: V4901权重 + 下行样本×1.5"""
        weights = super().compute_sample_weights(df, y)

        # 下行样本加权
        negative_mask = y < 0
        n_neg = negative_mask.sum()
        weights[negative_mask] *= self.DOWNSIDE_WEIGHT
        logger.info(f"    V4.9.0.2 下行加权: {n_neg:,} 负收益样本 × {self.DOWNSIDE_WEIGHT}")

        return weights

    def walk_forward_train(self, start_date=None, end_date=None,
                            purge_days=15, min_train_days=900,
                            val_days=120, test_days=120, step_days=90):
        """V4.9.0.2 Walk-Forward — 保存为v4902格式 + 自动生成WF摘要JSON"""
        import shutil
        import json as _json

        version_tag = 'v4902'
        version_str = 'v4.9.0.2'

        logger.info("=" * 60)
        logger.info(f"{version_str} Walk-Forward (V4.9.0.1 + Sharpe-Blend↑ + 下行加权)")
        logger.info("=" * 60)
        logger.info(f"  底座: V4.9.0.1 (61特征, Q95+trunc=10, 无头尾加权)")
        logger.info(f"  Sharpe-Blend: {self.TARGET_SHARPE_BLEND}")
        logger.info(f"  下行加权: {self.DOWNSIDE_WEIGHT}x")

        # 调用V485的walk_forward (与V4901相同)
        model_data, history = V485Trainer.walk_forward_train(
            self, start_date=start_date, end_date=end_date,
            purge_days=purge_days, min_train_days=min_train_days,
            val_days=val_days, test_days=test_days, step_days=step_days)

        if model_data.get('fast_check'):
            return model_data, history

        # 重命名 v485 → v4902
        v485_dir = PROJECT_ROOT / 'ml_models' / 'trained_models' / 'v485'
        out_dir = PROJECT_ROOT / 'ml_models' / 'trained_models' / version_tag
        out_dir.mkdir(parents=True, exist_ok=True)

        v485_files = sorted(v485_dir.glob('v485_*.pkl'), key=lambda f: f.stat().st_mtime)
        if v485_files:
            latest = v485_files[-1]
            timestamp = latest.stem.replace('v485_multi_target_', '')
            new_path = out_dir / f'{version_tag}_multi_target_{timestamp}.pkl'

            import joblib
            model_data['version'] = version_str
            model_data['v4902_innovations'] = {
                'base': 'V4.9.0.1 (no head-tail weighting)',
                'sharpe_blend': self.TARGET_SHARPE_BLEND,
                'downside_weight': self.DOWNSIDE_WEIGHT,
                'lambdarank': f'truncation={self.RANK_TRUNCATION}, grades={self.RANK_GRADES}',
                'quantile_model': f'LGB objective=quantile alpha={self.Q95_ALPHA}',
            }
            joblib.dump(model_data, new_path)
            logger.info(f"\n{version_str} model saved: {new_path}")

            # 复制辅助文件
            for aux in ['global_quantiles.npy', 'recommendation_thresholds.json']:
                src = v485_dir / aux
                if src.exists():
                    shutil.copy2(str(src), str(out_dir / aux))

            latest.unlink()
            for hf in v485_dir.glob(f'training_history_{timestamp}*'):
                hf.unlink()

            # 保存训练历史
            history['version'] = version_str
            history_path = out_dir / f'training_history_{timestamp}.json'
            with open(history_path, 'w', encoding='utf-8') as f:
                _json.dump(history, f, indent=2, ensure_ascii=False)
            latest_path = out_dir / 'training_history_latest.json'
            with open(latest_path, 'w', encoding='utf-8') as f:
                _json.dump(history, f, indent=2, ensure_ascii=False)

            # ═══ V5.1改动7: 保存WF摘要JSON ═══
            wf_summary = self._extract_wf_summary(history)
            if wf_summary:
                wf_path = out_dir / f'{version_tag}_wf_summary.json'
                with open(wf_path, 'w', encoding='utf-8') as f:
                    _json.dump(wf_summary, f, indent=2)
                logger.info(f"  WF摘要已保存: {wf_path}")
                logger.info(f"  WFER: {wf_summary.get('wfer', 'N/A')}")

        return model_data, history

    @staticmethod
    def _extract_wf_summary(history: dict) -> dict:
        """从训练历史中提取WF摘要 (供V5.1 WFER/OOS半衰期评估)"""
        wf_windows = history.get('walk_forward_windows', [])
        if not wf_windows:
            return {}

        is_sharpes = []
        oos_sharpes = []
        oos_monthly_ics = []

        for wf in wf_windows:
            # IS Sharpe
            is_metrics = wf.get('train_metrics', wf.get('is_metrics', {}))
            is_sharpe = is_metrics.get('sharpe_ratio', is_metrics.get('sharpe', None))
            if is_sharpe is not None:
                is_sharpes.append(float(is_sharpe))

            # OOS Sharpe
            oos_metrics = wf.get('test_metrics', wf.get('oos_metrics', {}))
            oos_sharpe = oos_metrics.get('sharpe_ratio', oos_metrics.get('sharpe', None))
            if oos_sharpe is not None:
                oos_sharpes.append(float(oos_sharpe))

            # OOS逐月IC (如果有)
            monthly_ic = oos_metrics.get('monthly_ic', [])
            if monthly_ic:
                oos_monthly_ics.append([float(ic) for ic in monthly_ic])

        result = {
            'is_sharpe': is_sharpes,
            'oos_sharpe': oos_sharpes,
        }

        if is_sharpes and oos_sharpes:
            is_mean = sum(is_sharpes) / len(is_sharpes)
            oos_mean = sum(oos_sharpes) / len(oos_sharpes)
            result['wfer'] = oos_mean / is_mean if is_mean > 0 else None

        if oos_monthly_ics:
            result['oos_monthly_ics'] = oos_monthly_ics

        return result
```
